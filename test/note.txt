package test

import (
	"os"
	"os/exec"
	"path/filepath"
	"regexp"
	"strings"

// TestLatexIncludeAndPackageIntegrity verifies that all \input{}, \include{},
// and custom local \usepackage{} directives resolve to actual files on disk.
func TestLatexIncludeAndPackageIntegrity(t *testing.T) {
	rootDir := ".."
	includeRegex := regexp.MustCompile(`\\(?:input|include)\{([^}]+)\}`)
	packageRegex := regexp.MustCompile(`\\usepackage(?:\[[^\]]*\])?\{([^}]+)\}`)

	// Custom local packages that must be present in the repository
	knownLocalPackages := map[string]bool{
		"fed-resume": true,
		"monserrat":  true, // Note single 't' matching repository file
	}

	err := filepath.Walk(rootDir, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		// Skip non-tex files, git directories, build caches, and the test fixture
		if info.IsDir() || filepath.Ext(path) != ".tex" || strings.Contains(path, "/test/test.tex") {
			return nil
		}
		if strings.Contains(path, "/autom4te.cache/") || strings.Contains(path, "/.git/") {
			return nil
		}

		data, err := os.ReadFile(path)
		if err != nil {
			t.Fatalf("Failed to read %s: %v", path, err)
		}
		content := string(data)
		dir := filepath.Dir(path)

		// 1. Validate \input{} and \include{}
		matches := includeRegex.FindAllStringSubmatch(content, -1)
		for _, m := range matches {
			target := strings.TrimSpace(m[1])

			// Try relative to current file directory, then relative to repo root
			candidates := []string{
				filepath.Join(dir, target),
				filepath.Join(dir, target+".tex"),
				filepath.Join(rootDir, target),
				filepath.Join(rootDir, target+".tex"),
			}

			found := false
			for _, candidate := range candidates {
				if _, err := os.Stat(candidate); err == nil {
					found = true
					break
				}
			}

			if !found {
				t.Errorf("Broken inclusion in %s: target '%s' not found", path, target)
			}
		}

		// 2. Validate custom local \usepackage{}
		pkgMatches := packageRegex.FindAllStringSubmatch(content, -1)
		for _, pm := range pkgMatches {
			pkgs := strings.Split(pm[1], ",")
			for _, pkg := range pkgs {
				pkgName := strings.TrimSpace(pkg)
				if knownLocalPackages[pkgName] {
					styName := pkgName + ".sty"
					styCandidates := []string{
						filepath.Join(dir, styName),
						filepath.Join(rootDir, "resume", styName),
						filepath.Join(rootDir, "biography", styName),
					}

					styFound := false
					for _, sc := range styCandidates {
						if _, err := os.Stat(sc); err == nil {
							styFound = true
							break
						}
					}
					if !styFound {
						t.Errorf("Missing local package in %s: %s not found on disk", path, styName)
					}
				}
			}
		}

		return nil
	})

	if err != nil {
		t.Fatalf("Walk failed: %v", err)
	}
}

package test

import (
	"bufio"
	"os"
	"path/filepath"
	"regexp"
	"strings"
	"testing"
)

// TestDateFormattingStandards validates date ranges line-by-line across resume sections
func TestDateFormattingStandards(t *testing.T) {
	// Pattern 1: Mon YYYY -- Present/YYYY
	// Pattern 2: YYYY -- YYYY/Present
	validDateRange := regexp.MustCompile(`(?:(?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\s+\d{4}|\d{4})\s*(?:--|–|-)\s*(?:Present|Current|\d{4}|(?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\s+\d{4})`)
	suspectDatePattern := regexp.MustCompile(`\b\d{4}\s*-\s*\d{4}\b|\b\d{2}/\d{4}\b`)

	targets := []string{
		filepath.Join("..", "resume", "sections", "experience.tex"),
		filepath.Join("..", "resume", "sections", "education.tex"),
		filepath.Join("..", "cv", "curriculum-vitae.md"),
	}

	for _, target := range targets {
		file, err := os.Open(target)
		if os.IsNotExist(err) {
			continue
		}
		if err != nil {
			t.Fatalf("Could not open %s: %v", target, err)
		}
		defer file.Close()

		scanner := bufio.NewScanner(file)
		lineNum := 0
		for scanner.Scan() {
			lineNum++
			line := strings.TrimSpace(scanner.Text())

			// If a line contains dates, ensure it adheres to the canonical delimiter style
			if suspectDatePattern.MatchString(line) && !validDateRange.MatchString(line) {
				t.Errorf("%s:%d: Non-standard date formatting detected: '%s'", target, lineNum, line)
			}
		}
	}
}

// TestStaleIdentityScan ensures no residual 'pale-shadow' accounts remain in active source files
func TestStaleIdentityScan(t *testing.T) {
	rootDir := ".."
	staleIdentifier := "pale-shadow"

	allowedExtensions := map[string]bool{
		".tex": true,
		".md":  true,
		".txt": true,
	}

	err := filepath.Walk(rootDir, func(path string, info os.FileInfo, err error) error {
		if err != nil || info.IsDir() {
			return nil
		}
		if !allowedExtensions[filepath.Ext(path)] {
			return nil
		}
		if strings.Contains(path, "/.git/") || strings.Contains(path, "/autom4te.cache/") {
			return nil
		}

		data, err := os.ReadFile(path)
		if err != nil {
			return nil
		}

		if strings.Contains(string(data), staleIdentifier) {
			t.Errorf("Stale identity '%s' found in %s", staleIdentifier, path)
		}
		return nil
	})

	if err != nil {
		t.Fatalf("Identity walk scan failed: %v", err)
	}
}

package test

import (
	"net/http"
	"net/http/httptest"
	"testing"
	"time"
)

// checkURLWithFallback replicates the core liveness logic: HEAD with GET fallback
func checkURLWithFallback(url string, client *http.Client) (int, error) {
	req, err := http.NewRequest(http.MethodHead, url, nil)
	if err != nil {
		return 0, err
	}
	req.Header.Set("User-Agent", "franklin-resume-test-checker/1.0")

	resp, err := client.Do(req)
	if err == nil {
		defer resp.Body.Close()
		if resp.StatusCode == http.StatusOK {
			return resp.StatusCode, nil
		}
	}

	// Fallback to GET on 405 Method Not Allowed or 403 Forbidden
	getReq, err := http.NewRequest(http.MethodGet, url, nil)
	if err != nil {
		return 0, err
	}
	getReq.Header.Set("User-Agent", "franklin-resume-test-checker/1.0")

	getResp, err := client.Do(getReq)
	if err != nil {
		return 0, err
	}
	defer getResp.Body.Close()
	return getResp.StatusCode, nil
}

func TestLinkCheckerFallbacksAndStatusHandling(t *testing.T) {
	// 1. Mock server that rejects HEAD (405) but honors GET (200)
	tsFallback := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == http.MethodHead {
			w.WriteHeader(http.StatusMethodNotAllowed)
			return
		}
		if r.Method == http.MethodGet {
			w.WriteHeader(http.StatusOK)
			return
		}
		w.WriteHeader(http.StatusBadRequest)
	}))
	defer tsFallback.Close()

	// 2. Mock server that returns 404
	tsNotFound := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusNotFound)
	}))
	defer tsNotFound.Close()

	client := &http.Client{Timeout: 3 * time.Second}

	// Test 1: Verify fallback mechanism recovers 405
	status, err := checkURLWithFallback(tsFallback.URL, client)
	if err != nil {
		t.Fatalf("Unexpected error for fallback endpoint: %v", err)
	}
	if status != http.StatusOK {
		t.Errorf("Expected fallback to return 200 OK, got %d", status)
	}

	// Test 2: Verify 404 is correctly classified
	status404, _ := checkURLWithFallback(tsNotFound.URL, client)
	if status404 != http.StatusNotFound {
		t.Errorf("Expected 404 status classification, got %d", status404)
	}
}

	CheckError(nil)
}

// TestCheckErrorFatal verifies that a non-nil error causes the process to exit/fatal.
// Run in a subprocess so it doesn't terminate the test runner.
func TestCheckErrorFatal(t *testing.T) {
	if os.Getenv("BE_CRASHING_PROCESS") == "1" {
		CheckError(errors.New("fatal sentinel error"))
		return
	}

	cmd := exec.Command(os.Args[0], "-test.run=TestCheckErrorFatal")
	cmd.Env = append(os.Environ(), "BE_CRASHING_PROCESS=1")
	err := cmd.Run()

	// The process must exit with non-zero exit status
	if e, ok := err.(*exec.ExitError); ok && !e.Success() {
		return
	}
	t.Fatalf("Process ran with err %v, expected non-zero exit status", err)
}
